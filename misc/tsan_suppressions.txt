# TSan: ThreadSanitizer
# https://github.com/google/sanitizers/wiki/threadsanitizersuppressions
#
# This file describes a number of places where TSAN detects problems in CRuby.
# Many of these indicate bugs. Others are benign (ex. data races that can be
# replaced with relaxed atomic loads)
#
# Usage: see "Building with Thread Sanitizer" in doc/contributing/debugging_ruby.md
#   Configure with:
#     ./configure CC=gcc cflags='-fsanitize=thread -O1 -g' LDFLAGS=-fsanitize=thread
#   Build and run with:
#     export RUBY_MN_THREADS=-1
#     export TSAN_OPTIONS="suppressions=$(pwd)/misc/tsan_suppressions.txt:die_after_fork=0:exitcode=0:disable_coredump=0:handle_segv=0:report_signal_unsafe=0:allow_user_segv_handler=1"
#
# Other useful TSAN_OPTIONS:
#   * halt_on_error=1
#   * strip_path_prefix=$(pwd)/

# vm->ractor.sched.grq_cnt++
race_top:ractor_sched_enq
race_top:ractor_sched_deq

# Race between vm_remove_ractor writing ractor count and
# native_thread_check_and_create_shared reading it during thread creation.
# The write happens when a ractor thread exits, the read happens when
# checking if new shared threads need to be created.
race:vm_remove_ractor

# th->sched.finished at end of co_start
race_top:rb_thread_sched_mark_zombies

# Races against timer thread setting th->sched.waiting_reason.flags
race_top:thread_sched_wait_events

# At thread start
race_top:rb_ractor_set_current_ec_

# TSan reports a lock-order-inversion between thread_sched_lock_ and this lock.
# It's unclear if that can cause a deadlock since the lock is on self
deadlock:ractor_lock_self

# TSan reports a deadlock when reacquiring the this lock after a barrier, but
# we know the other threads have been stopped
deadlock:rb_ractor_sched_barrier_start

# BUG: A Ractor's GC ages its objects (RVALUE_AGE_SET sets or clears
# FL_PROMOTED) with a plain read-modify-write of their flags, shareable ones
# too, while other Ractors read and write the flags of those (coderange
# caching on a shared String, FL_SHAREABLE, method entry flags...), so either
# side can lose the other's update.  `race:` because the top frame is the
# inlined RB_FL_SET_RAW/RB_FL_UNSET_RAW.
race:RVALUE_AGE_SET

# Inline caches and call cache updates
# Multiple threads can race when updating shared call caches during method lookups
# and argument forwarding. These races involve reading/writing cd->cc fields.
race_top:vm_cc_call_set
race_top:vm_cc_class_check
race_top:vm_search_cc
race_top:vm_search_method_slowpath0
race_top:rb_vm_opt_getconstant_path
race_top:vm_ic_attr_index_set
race:vm_ic_update
race:vm_caller_setup_fwd_args

# Race in shape_get_next where multiple threads simultaneously access and modify
# RCLASS_MAX_IV_COUNT and RCLASS_VARIATION_COUNT fields in class objects.
# One thread reads the field while another thread calls RCLASS_SET_MAX_IV_COUNT.
# This happens during instance variable shape transitions in multi-threaded code.
race:shape_get_next

# The races below were found getting test-all and btest to run under TSan.
# Entries marked BUG are real races that still need fixing: they are
# suppressed so that new reports stand out, not because they are harmless.
# Prefer race_top: a race: entry also matches every caller, allocation and
# thread creation stack in a report.

# BUG: Call caches, method entries and constant caches are created by one
# Ractor and published into call data and tables shared with other Ractors
# with plain stores (cd->cc = cc, vm_ccs_push...), which those read with plain
# loads, so nothing orders the initialization of the cache before its
# publication.  Likewise vm_cc_bf_set fills in a shared cache after it was
# published.
race_top:vm_cc_hit_p
race_top:vm_cc_cme
race_top:vm_cc_valid
race_top:vm_ic_hit_p
race_top:vm_call_single_noarg_leaf_builtin
race_top:vm_method_cfunc_entry
race_top:callable_method_entry_or_negative
race_top:vm_cc_bf_set
race_top:METHOD_ENTRY_CACHED_SET

# BUG: A Ractor's GC marks call caches in iseqs and classes shared with other
# Ractors, which write them at the same time.
race_top:cc_is_active
race_top:rb_iseq_mark_and_move

# BUG: rb_gc_single_objspace_p reads the Ractor count and the zombie objspace
# state without the VM lock, relying on the order they are written in (see
# vm_remove_ractor), but they are written with plain stores, which neither
# the compiler nor the CPU has to keep in order.
race_top:rb_gc_single_objspace_p

# BUG: Inheriting a dead Ractor's objspace (objspace_absorb) rewrites
# page->objspace of its pages while other Ractors look up the owner of objects
# on them.
race_top:gc_foreign_object_p

# BUG: A new native thread sets its Ractor's running_ec to its own ec in
# ruby_thread_set_native (from nt_start) before it has the GVL, so while the
# thread holding the GVL reads it in rb_ractor_thread_switch, running_ec can
# name a thread that is not running.
race_top:rb_ractor_thread_switch

# BUG: ARGF's line numbers ($.) are process-global and set by every Ractor
# reading lines.
race_top:rb_io_getline_1

# Two Ractors terminating the same frozen shared String (ENV[] with the same
# key, say) write the same byte.
race_top:str_fill_term

# Closing an IO resets fptr->fd while a thread blocked on it without the GVL
# reads it; the close then waits for the blocked threads to be done.
race_top:fptr_finalize_flush

# GC.stress= in one Ractor while others allocate.
race_top:rb_gc_impl_stress_set

# Creating a Fiber copies the running ec, interrupt_flag included, which the
# timer thread may be setting a timer interrupt in.
race_top:cont_save_thread

# ubf_set takes th->interrupt_lock while holding the sched lock, and
# threadptr_set_interrupt calls th->unblock.func (ubf_waiting), which takes
# the sched lock, with th->interrupt_lock held.  They cannot deadlock: ubf_set
# is only called with th->unblock.func NULL, so no ubf_waiting for th can be
# running then.
deadlock:ubf_set

race_top:find_last_event
